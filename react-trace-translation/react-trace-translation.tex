\section{React-tRace Core}

\subsection{Program-Level Syntax}

Throughout, constructs which only appear in intermediate stages 
of evaluation are in \grey{grey}.
Other non-standard/noteworthy constructs are \hl{highlighted}.
\begin{align*}
\text{Value } v &::= x \mid () \mid s \mid \lambda x.e \mid [] \mid v :: v\\
&\quad\mid \mhl{\C} \mid \mhl{\vtxt{v}} \mid \mhl{\vtag{v}{v}}
\mid \mhl{\vattr{v}{v}} \mid \mhl{\vclick{v}}\\
&\quad\mid \mgrey{\pair{\C}{v}} \mid \mgrey{\vset{\ell}{p}}\\
\text{Expression } e &::= \eret{v} \mid \elet{x}{e}{e} \mid v\ v \\
&\quad\mid \ecase{v}{[]}{e_1}{x_\mf{hd} :: x_\mf{tl}}{e_2}\\
\text{Body }\hat{e} &::= e \mid \mhl{\estate{x}{\ell}{v}{\hat{e}}} \mid \elet{x}{e}{\hat{e}}\\
\text{Program } P &::= e \mid \elet{x}{e}{P} \mid \mhl{\elet{\C(x)}{\hat{e}}{P}}\\
\text{Type } \ptp &::= \tone \mid \tstr \mid \ptp \arrow{E} \ptp
\mid [\ptp] \\
&\quad\mid \mhl{\tview} \mid \mhl{\tattr}
\mid \mhl{\tcomp{\ptp}} \mid \mhl{\tset{\ptp}}\\
\text{Effect } E &::= \epsilon \mid \mhl{\emut}
\end{align*}

\emph{\tb{Internal Representations}}.
\begin{align*}
\text{Boolean flag }b &::= \tt \mid \ff\\
\text{Type context }\Gamma &::= \map{x}{\ptp}\\
\text{Label set }\Sigma &::=\{\num{\ell}\}\\
\text{Component set }\Omega &::=\{\num{\C}\}\\
\text{Path context }\Pi &::= \map{p}{\{
    \mf{spec}: \C, \mf{labs}: \Sigma, \mf{tree}: b
\}}\\
\text{Component context }\Delta &::= \map{\C}{\pair{\ptp}{\map{\ell}{\ptp}}}\\
\text{Phase }\phi &::= \mf{Init} \mid \mf{Succ}\\
\text{Realized component }\pi &::= \{\mf{spec}: \pair{\C}{v}, 
\mf{st}: \map{\ell}{v}, \mf{tree}: u, \mf{rerender}: b\}\\
\text{Tree memory }m &::= \map{p}{\pi}\\
\text{Component definitions }\D &::= \map{\C}{x.\hat{e}}
\end{align*}
We elide the grammar of "runtime values" $u$,
which are distinct from normal values $v$, until a later section.

\emph{\tb{Notation}}.

We overload indexing notation on $\Delta$ for brevity,
i.e. $\Delta[\C]=\ptp$ and $\Delta[\C][\ell]=\ptp$.

We abbreviate $\Pi[p].\mf{spec}$ as simply $\Pi[p]$.

We use the notation $m \mid x = v$
to concisely denote nested map/record updates.
The above should be read as ``$m$ with field $x$ set to $v$''.
For instance, $m \mid m[p].\mf{st}[\ell] = v$ updates the
state field of $m$ at path $p$ and location $\ell$ to value $v$.

\emph{\tb{Ordering}}.

We define $\Pi \po \Pi'$ to hold when $\dom(\Pi) \subseteq \dom(\Pi')$
and, for all $p \in \mf{dom}(\Pi)$: \begin{itemize}
\item $\Pi[p] = \Pi'[p]$,
\item $\Pi[p].\mf{labs} \subseteq \Pi'[p].\mf{labs}$, and
\item if $\Pi[p].\mf{tree} = \mf{tt}$, then $\Pi'[p].\mf{tree} = \mf{tt}$.
\end{itemize}

$\po$ on all other contexts is given by standard subset inclusion.

\subsection{Program-Level Typing}

\subsubsection{Value Typing}

Judgment: $\tval{\Gamma}{\Pi}{v}{\ptp}$
\begin{mathpar}
\inferrule*[right=TV-Var] {
  \Gamma[x] = \ptp
} {
  \tval{\Gamma}{\Pi}{x}{\ptp}
}
\and
\inferrule*[right=TV-Unit] { } {
  \tval{\Gamma}{\Pi}{()}{\tone}
}
\and
\inferrule*[right=TV-Str] { } {
  \tval{\Gamma}{\Pi}{s}{\tstr}
}
\and
\inferrule*[right=TV-Lam] {
  \texp{\Gamma[x \mapsto \ptp_1]}{\Pi}{e}{\ptp_2}{E}
} {
  \tval{\Gamma}{\Pi}{\lambda x.e}{\ptp_1 \arrow{E} \ptp_2}
}
\and
\inferrule*[right=TV-Nil] { } {
  \tval{\Gamma}{\Pi}{[]}{[\ptp]}
}
\and
\inferrule*[right=TV-Cons] {
  \tval{\Gamma}{\Pi}{v_1}{\ptp}
  \and
  \tval{\Gamma}{\Pi}{v_2}{[\ptp]}
} {
  \tval{\Gamma}{\Pi}{v_1 :: v_2}{[\ptp]}
}
\and
\inferrule*[right=TV-Comp] {
  \Delta[\C] = \ptp
} {
  \tval{\Gamma}{\Pi}{\C}{\mf{Comp}\ \ptp}
}
\and
\inferrule*[right=TV-Text] {
  \tval{\Gamma}{\Pi}{v}{\tstr}
} {
  \tval{\Gamma}{\Pi}{\vtxt{v}}{\mf{View}}
}
\and
\inferrule*[right=TV-Tag] {
  \tval{\Gamma}{\Pi}{v_1}{[\mf{Attr}]}
  \and
  \tval{\Gamma}{\Pi}{v_2}{[\mf{View}]}
} {
  \tval{\Gamma}{\Pi}{\vtag{v_1}{v_2}}{\mf{View}}
}
\and
\inferrule*[right=TV-Attr] {
  \tval{\Gamma}{\Pi}{v_1}{\tstr}
  \and
  \tval{\Gamma}{\Pi}{v_2}{\tstr}
} {
  \tval{\Gamma}{\Pi}{\vattr{v_1}{v_2}}{\mf{Attr}}
}
\and
\inferrule*[right=TV-Click] {
  \tval{\Gamma}{\Pi}{v}{\tone \arrow{E} \tone}
} {
  \tval{\Gamma}{\Pi}{\vclick{v}}{\mf{Attr}}
}
\and
\inferrule*[right=\grey{TV-Spec}] {
  \tval{\Gamma}{\Pi}{v}{\Delta[\C]}
} {
  \tval{\Gamma}{\Pi}{\pair{\C}{v}}{\mf{View}}
}
\and
\inferrule*[right=\grey{TV-Set}] {
  \Delta[\Pi[p]][\ell] = \ptp
  \and
  \ell \in \Pi[p].\mf{labs}
} {
  \tval{\Gamma}{\Pi}{\vset{\ell}{p}}{\tset{\ptp}}
}
\end{mathpar}

Program judgment: $\tprog{\Gamma}{P}{\Omega}$
\begin{mathpar}
\inferrule*[right=TP-Lift] {
  \texp{\Gamma}{\cdot}{e}{\tview}{\epsilon}
} {
  \tprog{\Gamma}{e}{\emptyset}

\and
\inferrule*[right=TP-Let] {
  \texp{\Gamma}{\cdot}{e}{\ptp}{\epsilon}

  \tprog{\Gamma}{\elet{\C(x)}{\hat{e}}{P}}{\Omega \cup \{\C\}}

\end{mathpar}

\subsubsection{Memory Typing}

Define $\tmem{\Gamma}{\Pi}{m}$ to hold if $\dom(\Pi) = \dom(m)$
and, for all $p \in m$ (where $m[p].\mf{spec} = \pair{\C}{v}$):
\begin{itemize}
\item $\Pi[p] = \C$,
\item $\tval{\Gamma}{\Pi}{v}{\Delta[\C]}$,
\item $\dom(m[p].\mf{st}) = \Pi[p].\mf{labs}$,
\item $\forall \ell \in \dom(m[p].\mf{st})$,
$\tval{\Gamma}{\Pi}{m[p].\mf{st}[\ell]}{\Delta[\C][\ell]}$, and
\item if $\Pi[p].\mf{tree} = \tt$, then 
$\trhos{\Gamma}{\Pi}{m[p].\mf{tree}}{\ttree}$.
\end{itemize}

(We have not defined the tree-typing judgment yet.
We delay its introduction until the runtime section,
since it is unnecessary for the program-level proofs.)

\subsubsection{Component Definition Typing}

Define $\tdef{\Gamma}{\Omega}{\D}$ to hold if $\dom(\D)=\Omega$ and,
for all $\C \in \D$ (where $\D[\C] = x.\hat{e}$),
we have $\tbod{\Gamma[x \mapsto \Delta[\C]]}{\cdot}{\hat{e}}{\C}{\dom(\Delta[\C])}$.

\subsection{Program-Level Proofs}

Throughout, \emph{propositions} refer to (as-of-yet) unproved assumptions.

I denote \emph{deductions via inversion} with a label above the conclusion,
e.g.:
\begin{mathpar}
\inferrule* {(1)} {\tn{some judgment}}
\end{mathpar}

Comparatively, I denote the \emph{use of deductions} in a top-down derivation
with a label \emph{besides} the rule, e.g.:
\begin{mathpar}
\inferrule*[right=(1)] { } {\tn{some judgment}}
\end{mathpar}

\subsubsection{General Lemmas}

\begin{proposition}[Canonical forms.]
\label{thm:canon}
If $\tval{\cdot}{\Pi}{v}{\ptp}$, then:
\begin{itemize}
\item if $\ptp=\tone$, then $v=()$.
\item if $\ptp=\tstr$, then $v=s$.
\item if $\ptp=\ptp_1 \arrow{E} \ptp_2$, then $v=\lambda x.e$.
\item if $\ptp=[\ptp']$, then either $v=[]$ or $v = v_1 :: v_2$.
\item if $\ptp=\tview$, then either $v=\vtxt{v'}$,
    $v=\vtag{v_1}{v_2}$, or $v=\pair{\C}{v'}$.
\item if $\ptp=\tattr$, then either $v=\vattr{v_1}{v_2}$ 
    or $v=\vclick{v'}$.
\item if $\ptp=\tcomp{\ptp'}$, then $v=\C$.
\item if $\ptp=\tset{\ptp'}$, then $v=\vset{\ell}{p}$.
\end{itemize}
\end{proposition}

\begin{proposition}[Variable context weakening, values.]
\label{thm:wkgav}
If $\tval{\Gamma}{\Pi}{v}{\ptp}$ and $\Gamma \po \Gamma'$,
then $\tval{\Gamma'}{\Pi}{v}{\ptp}$.
\end{proposition}

\begin{proposition}[Path context weakening, values.]
\label{thm:wkpiv}
If $\tval{\Gamma}{\Pi}{v}{\ptp}$ and $\Pi \po \Pi'$,
then $\tval{\Gamma}{\Pi'}{v}{\ptp}$.
\end{proposition}

\begin{proposition}[Path context weakening, bodies.]
\label{thm:wkpib}
If $\tbod{\Gamma}{\Pi}{\hat{e}}{\C}{\Sigma}$ and $\Pi \po \Pi'$,
then $\tbod{\Gamma}{\Pi'}{\hat{e}}{\C}{\Sigma}$.
\end{proposition}

\begin{proposition}[Substitution lemma, expressions.]
\label{thm:subexp}
If $\texp{\Gamma[x \mapsto \ptp_1]}{\Pi}{e}{\ptp_2}{E}$
and $\tval{\Gamma}{\Pi}{v}{\ptp_1}$,
then $\texp{\Gamma}{\Pi}{e[v/x]}{\ptp_2}{E}$.
\end{proposition}

\begin{proposition}[Substitution lemma, bodies.]
\label{thm:subbod}
If we have $\tbod{\Gamma[x \mapsto \ptp]}{\Pi}{\hat{e}}{\C}{\Sigma}$
and $\tval{\Gamma}{\Pi}{v}{\ptp}$,
then $\tbod{\Gamma}{\Pi}{\hat{e}[v/x]}{\C}{\Sigma}$.
\end{proposition}

\begin{proposition}[Substitution lemma, programs.]
\label{thm:subprog}
If $\tprog{\Gamma[x \mapsto \ptp]}{P}{\Omega}$
and $\tval{\Gamma}{\cdot}{v}{\ptp}$,
then $\tprog{\Gamma}{P[v/x]}{\Omega}$.
\end{proposition}

\subsubsection{Expressions}

\begin{lemma}[Progress of expressions.]
\label{thm:progexp}
If $\texp{\cdot}{\Pi}{e}{\ptp}{E}$ and $\tmem{\cdot}{\Pi}{m}$,
then either:
\begin{enumerate}[label=(\roman*)]
\item $e = \eret{v}$, or
\item there exist some $e'$ and $m'$ such that
$\estep{m}{e}{e'}{m'}$.
\end{enumerate}

\begin{Proof}
By induction on the derivation of the expression typing judgment.

\tb{Case} \ts{TE-Ret}: We have 
$\texp{\cdot}{\Pi}{\eret{v}}{\ptp}{\epsilon}$,
and the theorem holds trivially by case \emph{(i)}.

\tb{Case} \ts{TE-Let}: \begin{mathpar}
\inferrule*[right=TE-Let] {
  \inferrule* {(1)} {\texp{\cdot}{\Pi}{e_1}{\ptp_1}{E}}
  \and
  \texp{[x \mapsto \ptp_1]}{\Pi}{e_2}{\ptp_2}{E}
} {
  \texp{\cdot}{\Pi}{\elet{x}{e_1}{e_2}}{\ptp_2}{E}
}
\end{mathpar}
\begin{itemize}
\item By inductive hypothesis on (1), we have two subcases on $e_1$:
either (a) $e_1 = \eret{v}$,
or (b) we have $\estep{m}{e_1}{e_1'}{m'}$.
\item In subcase (a), \ts{E-Ret} gives 
$\estep{m}{\elet{x}{\eret{v}}{e_2}}{e_2[v/x]}{m}$.
\item In subcase (b), we derive a step as follows:
\begin{mathpar}
\inferrule*[right=E-Let] {
  \inferrule*[right=(\tn{b})] { } {\estep{m}{e_1}{e_1'}{m'}}
} {
  \estep{m}{\elet{x}{e_1}{e_2}}{\elet{x}{e_1'}{e_2}}{m'}
}
\end{mathpar}
\item Thus, in any subcase, $\elet{x}{e_1}{e_2}$ steps,
and we satisfy conclusion \emph{(ii)}.
\end{itemize}

\tb{Case} \ts{TE-AppL}: \begin{mathpar}
\inferrule*[right=TE-AppL] {
  \inferrule* {(1)} {\tval{\Gamma}{\Pi}{v_1}{\ptp_1 \arrow{E} \ptp_2}}
  \and
  \tval{\Gamma}{\Pi}{v_2}{\ptp_1}
} {
  \texp{\Gamma}{\Pi}{v_1\ v_2}{\ptp_2}{E}
}
\end{mathpar}\begin{itemize}
\item By \thref{thm:canon} (canonical forms) on (1), $v_1=\lambda x.e$.
\item We apply \ts{E-AppL} to get that
$\estep{m}{(\lambda x. e)\ v_2}{e[v_2/x]}{m}$,
satisfying conclusion \emph{(ii)}.
\end{itemize}

\tb{Case} \ts{TE-AppC}: \begin{mathpar}
\inferrule*[right=TE-AppC] {
  \inferrule* {(1)} {\tval{\Gamma}{\Pi}{v_1}{\tcomp{\ptp}}}
  \and
  \tval{\Gamma}{\Pi}{v_2}{\ptp}
} {
  \texp{\Gamma}{\Pi}{v_1\ v_2}{\tview}{\epsilon}
}
\end{mathpar}\begin{itemize}
\item By \thref{thm:canon} (canonical forms) on (1), $v_1 = \C$.
\item \ts{E-AppC} gives
$\estep{m}{\C\ v_2}{\eret{\pair{\C}{v_2}}}{m}$,
satisfying conclusion \emph{(ii)}.
\end{itemize}

\tb{Case} \ts{TE-AppS}: \begin{mathpar}
\inferrule*[right=TE-AppS] {
  \inferrule* {(1)} {\tval{\Gamma}{\Pi}{v_1}{\tset{\ptp}}}
  \and
  \tval{\Gamma}{\Pi}{v_2}{\ptp}
} {
  \texp{\Gamma}{\Pi}{v_1\ v_2}{\tone}{\emut}
}
\end{mathpar}\begin{itemize}
\item By \thref{thm:canon} (canonical forms) on (1), $v_1 = \vset{\ell}{p}$.
\item We apply \ts{E-AppS} to get that
$\estep{m}{\vset{\ell}{p}\ v}{\eret{()}}{m'}$,
where we define $m' = m \mid m[p].\mf{st}[\ell] = v_2$ 
and $m[p].\mf{rerender} = \tt$,
satisfying conclusion \emph{(ii)}.
\end{itemize}

\tb{Case} \ts{TE-Case}: \begin{mathpar}
\inferrule*[right=TE-Case] {
  \inferrule* {(1)} {\tval{\Gamma}{\Pi}{v}{[\ptp_1]}}
  \and
  \texp{\Gamma}{\Pi}{e_1}{\ptp_2}{E}
  \\
  \texp{\Gamma[x_\mf{hd} \mapsto \ptp_1, x_\mf{tl} \mapsto [\ptp_1]]}
  {\Pi}{e_2}{\ptp_2}{E}
} {
  \texp{\Gamma}{\Pi}{\ecase{v}{[]}{e_1}{x_\mf{hd} :: x_\mf{tl}}{e_2}}{\ptp_2}{E}
}
\end{mathpar}
\begin{itemize}
\item By \thref{thm:canon} (canonical forms) on (1), either 
(a) $v=[]$ or (b) $v=v_\mf{hd} :: v_\mf{tl}$.
\item (a) If $v=[]$, then \ts{E-CaseN} gives\\
$\estep{m}{\ecase{[]}{[]}{e_1}{x_\mf{h} :: x_\mf{t}}{e_2}}{e_1}{m}$.
\item (b) If $v=v_\mf{hd} :: v_\mf{tl}$, then \ts{E-CaseC} gives\\
$\estep{m}{\ecase{v_\mf{hd} :: v_\mf{tl}}{[]}{e_1}{x_\mf{hd} :: x_\mf{tl}}{e_2}}
{e_2[v_\mf{hd} / x_\mf{hd}, v_\mf{tl} / x_\mf{tl}]}{m}$.
\item In either subcase, we satisfy conclusion \emph{(ii)}.
\end{itemize}

\tb{Case} \ts{TE-Sub}: \begin{mathpar}
\inferrule*[right=TE-Sub] {
  \inferrule* {(1)} {\texp{\Gamma}{\Pi}{e}{\ptp}{\epsilon}}
} {
  \texp{\Gamma}{\Pi}{e}{\ptp}{\emut}
}
\end{mathpar}
The result follows immediately from the inductive hypothesis,
applied to the (more conservative) premise (1).
\end{Proof}
\end{lemma}

\begin{lemma}[Preservation of expressions.]
\label{thm:prezexp}
If
\begin{enumerate}[label=(\roman*)]
\item $\texp{\Gamma}{\Pi}{e}{\ptp}{E}$,
\item $\tmem{\Gamma}{\Pi}{m}$, and
\item $\estep{m}{e}{e'}{m'}$ for some $e'$ and $m'$,
\end{enumerate}
then
\begin{enumerate}[label=(\roman*)]
\item $\texp{\Gamma}{\Pi}{e'}{\ptp}{E}$ and
\item $\tmem{\Gamma}{\Pi}{m'}$.
\end{enumerate}

\begin{Proof}
By induction on derivation of the step judgment \emph{(iii)}.

\tb{Case} \ts{E-Let}: We have \begin{mathpar}
\inferrule*[right=E-Let] {
  \inferrule* {(1)} {\estep{m}{e_1}{e_1'}{m'}}
} {
  \estep{m}{\elet{x}{e_1}{e_2}}{\elet{x}{e_1'}{e_2}}{m'}
}
\and
\inferrule*[right=TE-Let] {
  \inferrule* {(2)} {\texp{\Gamma}{\Pi}{e_1}{\ptp_1}{E}}
  \and
  \inferrule* {(3)} {\texp{\Gamma[x \mapsto \ptp_1]}{\Pi}{e_2}{\ptp_2}{E}}
} {
  \texp{\Gamma}{\Pi}{\elet{x}{e_1}{e_2}}{\ptp_2}{E}
}
\end{mathpar}
\begin{itemize}
\item We apply the inductive hypothesis with (1), (2), and premise \emph{(ii)}
to get $\Gamma \semi \Pi \vdm m'$, satisfying conclusion \emph{(ii)}, and
(4) $\Gamma \semi \Pi \vdexp e_1': \ptp_1 \mid E$.
\item We reapply \ts{TE-Let} as follows to get conclusion \emph{(i)}:
\begin{mathpar}
\inferrule*[right=TE-Let] {
  \inferrule*[right=(4)] { } {\texp{\Gamma}{\Pi}{e_1'}{\ptp_1}{E}}
  \and
  \inferrule*[right=(3)] { } {
    \texp{\Gamma[x \mapsto \ptp_1]}{\Pi}{e_2}{\ptp_2}{E}
  }
} {
  \texp{\Gamma}{\Pi}{\elet{x}{e_1'}{e_2}}{\ptp_2}{E}
}
\end{mathpar}
\end{itemize}

\tb{Case} \ts{E-Ret}: We have $\estep{m}{\elet{x}{\eret{v}}{e}}{e[v/x]}{m}$ and
\begin{mathpar}
\inferrule*[right=TE-Let] {
  \inferrule* {(1)} {\texp{\Gamma}{\Pi}{\eret{v}}{\ptp_1}{E}}
  \and
  \inferrule* {(2)} {\texp{\Gamma[x \mapsto \ptp_1]}{\Pi}{e_2}{\ptp_2}{E}}
} {
  \texp{\Gamma}{\Pi}{\elet{x}{\eret{v}}{e}}{\ptp_2}{E}
}
\end{mathpar}
\begin{itemize}
\item (1) lets us derive (3) $\tval{\Gamma}{\Pi}{v}{\ptp_1}$
by subcasing on the value of $E$:
\begin{mathpar}
\inferrule*[right=TE-Ret] {
  \inferrule* {(3)} {\tval{\Gamma}{\Pi}{v}{\ptp_1}}
} {
  \texp{\Gamma}{\Pi}{\eret{v}}{\ptp_1}{\epsilon}
}
\and
\inferrule*[right=TE-Sub] {
  \inferrule*[right=TE-Ret] {
    \inferrule* {(3)} {\tval{\Gamma}{\Pi}{v}{\ptp_1}}
  } {
    \texp{\Gamma}{\Pi}{\eret{v}}{\ptp_1}{\epsilon}
  }
} {
  \texp{\Gamma}{\Pi}{\eret{v}}{\ptp_1}{\emut}
}
\end{mathpar}
\item By \thref{thm:subexp} (substitution for expressions)
on (2) and (3), we get
$\texp{\Gamma}{\Pi}{e[v/x]}{\ptp_2}{E}$ and we are done.
\end{itemize}

\tb{Case} \ts{E-AppL}: We have $\estep{m}{(\lambda x. e)\ v}{e[v/x]}{m}$ and
\begin{mathpar}
\inferrule*[right=TE-AppL] {
  \inferrule*[right=TV-Lam] {
    \inferrule* {(1)} {\texp{\Gamma[x \mapsto \ptp_1]}{\Pi}{e}{\ptp_2}{E}}
  } {
    \tval{\Gamma}{\Pi}{\lambda x.e}{\ptp_1 \arrow{E} \ptp_2}
  }
  \and
  \inferrule* {(2)} {\tval{\Gamma}{\Pi}{v}{\ptp_1}}
} {
  \texp{\Gamma}{\Pi}{(\lambda x. e)\ v}{\ptp_2}{E}
}
\end{mathpar}
\begin{itemize}
\item By \thref{thm:subexp} (substitution for expressions)
on (1) and (2), we get
$\texp{\Gamma}{\Pi}{e[v/x]}{\ptp_2}{E}$ and we are done.
\end{itemize}

\tb{Case} \ts{E-AppC}: We have $\estep{m}{\C\ v}{\eret{\pair{\C}{v}}}{m}$ and
\begin{mathpar}
\inferrule*[right=TE-AppC] {
  \inferrule*[right=TV-Comp] {
    \inferrule* {(1)} {\Delta[\C] = \ptp}
  } {
    \tval{\Gamma}{\Pi}{\C}{\mf{Comp}\ \ptp}
  }
  \and
  \inferrule* {(2)} {\tval{\Gamma}{\Pi}{v}{\ptp}}
} {
  \texp{\Gamma}{\Pi}{\C\ v}{\tview}{\epsilon}
}
\end{mathpar}
\begin{itemize}
\item From (1) and (2) we get (3) that $\tval{\Gamma}{\Pi}{v}{\Delta[\C]}$.
We can then derive conclusion \emph{(i)}:
\begin{mathpar}
\inferrule*[right=TE-Ret] {
  \inferrule*[right=TV-Spec] {
    \inferrule* {(3)} {\tval{\Gamma}{\Pi}{v}{\Delta[\C]}}
  } {
    \tval{\Gamma}{\Pi}{\pair{\C}{v}}{\tview}
  }
} {
  \texp{\Gamma}{\Pi}{\eret{\pair{\C}{v}}}{\tview}{\epsilon}
}
\end{mathpar}
\end{itemize}

\tb{Case} \ts{E-AppS}: We have $\estep{m}{\vset{\ell}{p}\ v}{\eret{()}}{m'}$ and
\begin{mathpar}
\inferrule*[right=TE-AppS] {
  \inferrule*[right=TV-Set] {
    \inferrule* {(1)} {\Delta[\Pi[p]][\ell] = \ptp}
    \and
    \inferrule* {(2)} {\ell \in \Pi[p].\mf{labs}}
  } {
    \tval{\Gamma}{\Pi}{\vset{\ell}{p}}{\tset{\ptp}}
  }
  \
  \inferrule* {(3)} {\tval{\Gamma}{\Pi}{v}{\ptp}}
} {
  \texp{\Gamma}{\Pi}{\vset{\ell}{p}\ v}{\tone}{\emut}
}
\end{mathpar}
where $m' = m \mid m[p].\mf{st}[\ell] = v$ and $m[p].\mf{rerender} = \mf{tt}$.
\begin{itemize}
\item Conclusion \emph{(i)} follows from the derivation:
\begin{mathpar}
\inferrule*[right=TE-Sub] {
  \inferrule*[right=TE-Ret] {
    \inferrule*[right=TV-Unit] { } {
      \tval{\Gamma}{\Pi}{()}{\tone}
    } 
  } {
    \texp{\Gamma}{\Pi}{\eret{()}}{\tone}{\epsilon}
  }
} {
  \texp{\Gamma}{\Pi}{\eret{()}}{\tone}{\emut}
}
\end{mathpar}
\item To show $\tmem{\Gamma}{\Pi}{m'}$ given premise $\tmem{\Gamma}{\Pi}{m}$,
it suffices to show (\emph{ii}-a) that $\dom(m'[p].\mf{st}) = \Pi[p].\mf{labs}$
and (\emph{ii}-b) that 
$\tval{\Gamma}{\Pi}{m[p].\mf{st}[\ell]}{\Delta[\Pi[p]][\ell]}$.
\item (\emph{ii}-a) follows from (2) and definition of $m'$.
\item To show (\emph{ii}-b), by (1) and definition of $m'$,
we only need to show that $\tval{\Gamma}{\Pi}{v}{\ptp}$.
This is given by (3), and we are done.
\end{itemize}

\tb{Case} \ts{E-CaseN}: We have 
$\estep{m}{\ecase{[]}{[]}{e_1}{x_\mf{h} :: x_\mf{t}}{e_2}}{e_1}{m}$ and
\begin{mathpar}
\inferrule*[right=TE-Case] {
  \inferrule* {(1)} {\texp{\Gamma}{\Pi}{e_1}{\ptp_2}{E}} \and \cdots
} {
  \texp{\Gamma}{\Pi}{\ecase{[]}{[]}{e_1}{x_\mf{hd} :: x_\mf{tl}}{e_2}}{\ptp_2}{E}
}
\end{mathpar}
\begin{itemize}
\item The conclusion follows immediately from (1).
\end{itemize}

\tb{Case} \ts{E-CaseC}: We have\\
$\estep{m}{\ecase{v_\mf{h} :: v_\mf{t}}{[]}{e_1}{x_\mf{h} :: x_\mf{t}}{e_2}}
{e_2[v_\mf{h} / x_\mf{h}, v_\mf{t} / x_\mf{t}]}{m}$ and
\begin{mathpar}
\inferrule*[right=TE-Case] {
  \inferrule* {(1)} {\tval{\Gamma}{\Pi}{v_\mf{h} :: v_\mf{t}}{[\ptp_1]}}
  \and
  \inferrule* {(2)} {\texp{\Gamma'}{\Pi}{e_2}{\ptp_2}{E}}
  \and
  \cdots
} {
  \texp{\Gamma}{\Pi}{\ecase{v_\mf{h} :: v_\mf{t}}{[]}{e_1}{x_\mf{h} :: x_\mf{t}}{e_2}}
  {\ptp_2}{E}
}
\end{mathpar}
where $\Gamma' = \Gamma[x_\mf{h} \mapsto \ptp', x_\mf{t} \mapsto [\ptp']]$.
\begin{itemize}
\item We can further derive from (1):
\begin{mathpar}
\inferrule*[right=TV-Cons] {
  \inferrule* {(3)} {\Gamma \semi \Pi \vdval v_\mf{h}: \ptp'}
  \and
  \inferrule* {(4)} {\Gamma \semi \Pi \vdval v_\mf{t}: [\ptp']}
} {
  \Gamma \semi \Pi \vdval v_\mf{h} :: v_\mf{t} : [\ptp']\ (1)
}
\end{mathpar}
\item We apply \thref{thm:subexp} (substitution) twice to (2) 
using values (3) and (4) to get that 
$\Gamma \semi \Pi \vdval e_2[v_\mf{h} / x_\mf{h}, v_\mf{t} / x_\mf{t}]: \ptp \mid E$
and we are done.
\end{itemize}
\end{Proof}
\end{lemma}

The above two lemmas are sufficient for type soundness of expressions,
but we also prove a "purity lemma" on expressions for later use:

\begin{lemma}[Purity of expressions.]
\label{thm:pureexp}
If $\cdot \semi \Pi \vdexp e: \ptp \mid \epsilon$
and $m \semi e \mapsto e' \semi m'$,
then $m' = m$.
\begin{Proof}
By induction on the derivation of the typing judgment.

\tb{Case} \ts{TE-Ret}: We have \begin{mathpar}
\inferrule*[right=TE-Ret] {
  \tval{\cdot}{\Pi}{v}{\ptp}
} {
  \texp{\cdot}{\Pi}{\eret{v}}{\ptp}{\epsilon}
}
\end{mathpar}
But there is no rule that steps an expression of form $\eret{v}$,
so the theorem holds vacuously.

\tb{Case} \ts{TE-Let}: We have \begin{mathpar}
\and
\inferrule*[right=TE-Let] {
  \inferrule* {(1)} {\texp{\cdot}{\Pi}{e_1}{\ptp_1}{E}}
  \and
  \texp{[x \mapsto \ptp_1]}{\Pi}{e_2}{\ptp_2}{E}
} {
  \texp{\cdot}{\Pi}{\elet{x}{e_1}{e_2}}{\ptp_2}{E}
}
\end{mathpar}
\begin{itemize}
\item By \thref{thm:progexp} (progress) on (1), we either have (a) $e=\eret{v}$
or (b) $\estep{m}{e}{e'}{m'}$.
\item If (a) $e=\eret{v}$, then only \ts{E-Ret} applies, giving us
$\estep{m}{\elet{x}{\eret{v}}{e}}{e[v/x]}{m}$, and $m'=m$.
\item If instead (b) $\estep{m}{e}{e'}{m'}$, then we apply the inductive hypothesis
on (1) and (b), giving us again that $m'=m$.
\item Thus, in either case, the desired statement holds.
\end{itemize}

\tb{Case} \ts{TE-AppL}: We have
\begin{mathpar}
\inferrule*[right=TE-AppL] {
  \inferrule* {(1)} {\tval{\cdot}{\Pi}{v_1}{\ptp_1 \arrow{\epsilon} \ptp_2}}
  \and
  \tval{\cdot}{\Pi}{v_2}{\ptp_1}
} {
  \texp{\cdot}{\Pi}{v_1\ v_2}{\ptp_2}{\epsilon}
}
\end{mathpar}
\begin{itemize}
\item By \thref{thm:canon} (canonical forms) on (1), $v_1 = \lambda x.e$.
\item The only step rule which applies is \ts{E-AppL}, which says
$\estep{m}{(\lambda x. e)\ v_2}{e[v_2/x]}{m}$,
i.e. $m' = m$ and we are done.
\end{itemize}

\tb{Case} \ts{TE-AppC}: We have
\begin{mathpar}
\inferrule*[right=TE-AppC] {
  \inferrule* {(1)} {\tval{\cdot}{\Pi}{v_1}{\tcomp{\ptp}}}
  \and
  \tval{\cdot}{\Pi}{v_2}{\ptp}
} {
  \texp{\cdot}{\Pi}{v_1\ v_2}{\tview}{\epsilon}
}
\end{mathpar}
\begin{itemize}
\item By \thref{thm:canon} (canonical forms) on (1), $v_1 = \C$.
\item The only step rule that applies is \ts{E-AppC}, which says
that $\estep{m}{\C\ v}{\eret{\pair{\C}{v}}}{m}$,
i.e. $m' = m$ and we are done.
\end{itemize}

\tb{Case} \ts{TE-Case}: We have \begin{mathpar}
\inferrule*[right=TE-Case] {
  \inferrule* {(1)} {\tval{\cdot}{\Pi}{v}{[\ptp_1]}}
  \and \cdots
} {
  \texp{\cdot}{\Pi}{\ecase{v}{[]}{e_1}{x_\mf{hd} :: x_\mf{tl}}{e_2}}{\ptp_2}{E}
}
\end{mathpar}
\begin{itemize}
\item By \thref{thm:canon} (canonical forms) on (1), 
$v_1 = []$ or $v_1 = v_\mf{hd} :: v_\mf{tl}$.
\item If $v_1=[]$, then \ts{E-CaseN} says that\\
$\estep{m}{\ecase{[]}{[]}{e_1}{x_\mf{h} :: x_\mf{t}}{e_2}}{e_1}{m}$.
\item If $v_1=\mf{false}$, then \ts{E-CaseC} says that\\
$\estep{m}{\ecase{v_\mf{h} :: v_\mf{t}}{[]}{e_1}{x_\mf{h} :: x_\mf{t}}{e_2}}
{e_2[v_\mf{h} / x_\mf{h}, v_\mf{t} / x_\mf{t}]}{m}$.
\item In either case, $m' = m$ and we are done.
\end{itemize}

\tb{Cases} \ts{TE-AppS} and \ts{TE-Sub}
are both impossible, since they type their conclusions
with the $\mf{Mut}$ effect.
\end{Proof}
\end{lemma}

\subsubsection{Bodies}

\begin{lemma}[Progress of bodies, initial phase.]
\label{thm:progbodi}
If
$\tbod{\cdot}{\Pi}{\hat{e}}{\C}{\Sigma}$ and $\tmem{\cdot}{\Pi}{m}$,
then either:
\begin{enumerate}[label=(\roman*)]
\item $\hat{e} = \eret{v}$, or
\item there exist some $\hat{e}'$ and $m'$ such that
$\bstep{m}{\hat{e}}{\mf{Init}}{p}{\hat{e}'}{m'}$.
\end{enumerate}

\begin{Proof}
By induction on the derivation of the body typing judgment.

\tb{Case} \ts{TB-Lift}: \begin{mathpar}
\inferrule*[right=TB-Lift] {
  \inferrule* {(1)} {\texp{\cdot}{\Pi}{e}{\tview}{\epsilon}}
} {
  \tbod{\cdot}{\Pi}{e}{\C}{\emptyset}
}
\end{mathpar}
\begin{itemize}
\item By \thref{thm:progexp} (expression progress) on (1), we have two subcases:
either $e=\eret{v}$ (and we are done by conclusion (\emph{i})), 
or we have some $e'$ and $m'$ such that
(2) $\estep{m}{e}{e'}{m'}$.
\item By \thref{thm:pureexp} (purity) on (1) and (2), we get (3) that
$\estep{m}{e}{e'}{m}$.
\item We derive the conclusion as follows:
\begin{mathpar}
\inferrule*[right=EB-Lift] {
  \inferrule*[right=(3)] { } {\estep{m}{e}{e'}{m}}
} {
  \bstep{m}{e}{\mf{Init}}{p}{e'}{m}
}
\end{mathpar}
\end{itemize}

\tb{Case} \ts{TB-Let}: \begin{mathpar}
\inferrule*[right=TB-Let] {
  \inferrule* {(1)} {\texp{\cdot}{\Pi}{e}{\ptp}{\epsilon}}
  \and
  \tbod{[x \mapsto \ptp]}{\Pi}{\hat{e}}{\C}{\Sigma}
} {
  \tbod{\cdot}{\Pi}{\elet{x}{e}{\hat{e}}}{\C}{\Sigma}
}
\end{mathpar}\begin{itemize}
\item By \thref{thm:progexp} (expression progress) on (1), we have two subcases:
either (a) $e = \eret{v}$,
or (b) we have $\estep{m}{e}{e'}{m'}$.
\item In subcase (a), \ts{EB-Ret} gives
$\bstep{m}{\elet{x}{\eret{v}}{\hat{e}}}{\mf{Init}}{p}{\hat{e}[v/x]}{m}$.
\item In subcase (b), by \thref{thm:pureexp} (purity) on (1) and (b),
we get (3) that $\estep{m}{e}{e'}{m}$.
We derive a step as follows:
\begin{mathpar}
\inferrule*[right=EB-Let] {
  \inferrule*[right=(3)] { } {\estep{m}{e}{e'}{m}}
} {
  \bstep{m}{\elet{x}{e}{\hat{e}}}{\mf{Init}}{p}{\elet{x}{e'}{\hat{e}}}{m}
}
\end{mathpar}
\item Thus, in either subcase, we take a step and satisfy conclusion \emph{(ii)}.
\end{itemize}

\tb{Case} \ts{TB-State}: \begin{mathpar}
\inferrule*[right=TB-State] {
  \Delta[\C][\ell] = \ptp
  \and
  \tval{\cdot}{\Pi}{v}{\ptp}
  \\
  \tbod{[x \mapsto \ptp, x_\mf{set} \mapsto \tset{\ptp}]}
  {\Pi}{\hat{e}}{\C}{\Sigma}
} {

}
\end{mathpar}
\begin{itemize}
\item Rule \ts{EB-StI} immediately applies, giving us

{\hat{e}[v/x, \vset{\ell}{p}/x_\mf{set}]}{m'}$$
for $m' = m \mid m[p].\mf{st}[\ell] = v$.
\end{itemize}
\end{Proof}
\end{lemma}

\begin{lemma}[Preservation of bodies, initial phase.]
\label{thm:prezbodi}
If
\begin{enumerate}[label=(\roman*)]
\item $\tbod{\Gamma}{\Pi}{\hat{e}}{\C}{\Sigma}$,
\item $\tmem{\Gamma}{\Pi}{m}$, and
\item $\bstep{m}{\hat{e}}{\mf{Init}}{p}{\hat{e}'}{m'}$,
\end{enumerate}
for $\C=\Pi[p]$,
then there exist some $\Pi' \rpo \Pi$ and $\Sigma' \subseteq \Sigma$ such that:
\begin{enumerate}[label=(\roman*)]
\item $\tbod{\Gamma}{\Pi'}{\hat{e}'}{\C}{\Sigma'}$,
\item $\tmem{\Gamma}{\Pi'}{m'}$, and
\item $\Pi'[p].\mf{labs} \cup \Sigma' = \Pi[p].\mf{labs} \cup \Sigma$.
\end{enumerate}

\begin{Proof}
Unless otherwise specified, we choose $\Pi' = \Pi$ and $\Sigma' = \Sigma$
such that conclusions \emph{(ii)} and \emph{(iii)} are trivial.

We proceed by induction on derivation of the step judgment \emph{(iii)}.

\tb{Case} \ts{EB-Lift}: We have \begin{mathpar}
\inferrule*[right=EB-Lift] {
  \inferrule* {(1)} {\estep{m}{e}{e'}{m}}
} {
  \bstep{m}{e}{\mf{Init}}{p}{e'}{m}
}
\and
\inferrule*[right=TB-Lift] {
  \inferrule* {(2)} {\texp{\Gamma}{\Pi}{e}{\tview}{\epsilon}}
} {
  \tbod{\Gamma}{\Pi}{e}{\C}{\emptyset}
}
\end{mathpar}
\begin{itemize}
\item By \thref{thm:prezexp} (expression preservation)
on (1), (2), and premise \emph{(ii)},
we get (3) $\texp{\Gamma}{\Pi}{e'}{\tview}{\epsilon}$.
\item We satisfy conclusion \emph{(i)} via the following derivation:
\begin{mathpar}
\inferrule*[right=TB-Lift] {
  \inferrule*[right=(3)] { } {\texp{\Gamma}{\Pi}{e'}{\tview}{\epsilon}}
} {
  \tbod{\Gamma}{\Pi}{e'}{\C}{\emptyset}
}
\end{mathpar}
\end{itemize}

\tb{Case} \ts{EB-Let}: We have \begin{mathpar}
\inferrule*[right=EB-Let] {
  \inferrule* {(1)} {\estep{m}{e}{e'}{m}}
} {
  \bstep{m}{\elet{x}{e}{\hat{e}}}{\mf{Init}}{p}{\elet{x}{e'}{\hat{e}}}{m}
}
\and
\inferrule*[right=TB-Let] {
  \inferrule* {(2)} {\texp{\Gamma}{\Pi}{e}{\ptp}{\epsilon}}
  \and
  \inferrule* {(3)} {\tbod{\Gamma[x \mapsto \ptp]}{\Pi}{\hat{e}}{\C}{\Sigma}}
} {
  \tbod{\Gamma}{\Pi}{\elet{x}{e}{\hat{e}}}{\C}{\Sigma}
}
\end{mathpar}\begin{itemize}
\item By \thref{thm:prezexp} (expression preservation)
on (1), (2), and premise \emph{(ii)},
we get that (4) $\Gamma \semi \Pi \vdexp e': \ptp \mid \epsilon$.
\item We derive conclusion \emph{(i)} as follows:
\begin{mathpar}
\inferrule*[right=TB-Let] {
  \inferrule*[right=(4)] { } {\texp{\Gamma}{\Pi}{e'}{\ptp}{\epsilon}}
  \and
  \inferrule*[right=(3)] { } {\tbod{\Gamma[x \mapsto \ptp]}{\Pi}{\hat{e}}{\C}{\Sigma}}
} {
  \tbod{\Gamma}{\Pi}{\elet{x}{e'}{\hat{e}}}{\C}{\Sigma}
}
\end{mathpar}
\end{itemize}

\tb{Case} \ts{EB-Ret}: We have
(1) $\bstep{m}{\elet{x}{\eret{v}}{\hat{e}}}{\mf{Init}}{p}{\hat{e}[v/x]}{m}$
and:
\begin{mathpar}
\inferrule*[right=TB-Let] {
  \inferrule*[right=TE-Ret] {
    \inferrule* {(2)} {\tval{\Gamma}{\Pi}{v}{\ptp}}
  } {
    \texp{\Gamma}{\Pi}{\eret{v}}{\ptp}{\epsilon}
  }
  \
  \inferrule* {(3)} {\tbod{\Gamma[x \mapsto \ptp]}{\Pi}{\hat{e}}{\C}{\Sigma}}
} {
  \tbod{\Gamma}{\Pi}{\elet{x}{\eret{v}}{\hat{e}}}{\C}{\Sigma}
}
\end{mathpar}
\begin{itemize}
\item By \thref{thm:subbod} (substitution) on (2) and (3),
we get that $\tbod{\Gamma}{\Pi}{\hat{e}[v/x]}{\C}{\Sigma}$,
satisfying conclusion \emph{(i)}.
\end{itemize}

\tb{Case} \ts{EB-StI}: We have\\
$\bstep{m}{\estate{x}{\ell}{v}{\hat{e}}}{\mf{Init}}{p}
{\hat{e}[v/x, \vset{\ell}{p}/x_\mf{set}]}{m'}$\\
where $m' = m \mid m[p].\mf{st}[\ell] = v$, and:
\begin{mathpar}
\inferrule*[right=TB-State] {
  \Delta[\C][\ell] = \ptp \ (1)
  \and
  \tval{\Gamma}{\Pi}{v}{\ptp}\ (2)
  \\
  \tbod{\Gamma[x \mapsto \ptp, x_\mf{set} \mapsto \tset{\ptp}]}
  {\Pi}{\hat{e}}{\C}{\hat{\Sigma}}\ (3)
} {
  \tbod{\Gamma}{\Pi}{\estate{x}{\ell}{v}{\hat{e}}}{\C}{\hat{\Sigma} \cup \{\ell\}}
}
\end{mathpar}
\begin{itemize}
\item Choose $\Pi' = \Pi \mid \Pi[p].\mf{labs} = \Pi[p].\mf{labs} \cup \{\ell\}$
and $\Sigma' = \hat{\Sigma}$ (such that $\Sigma = \Sigma' \cup \{\ell\}$).
Clearly both $\Pi \po \Pi'$ and $\Sigma' \subseteq \Sigma$.
\item We derive the setter type as follows 
(recalling that we defined $\C=\Pi[p]$):
\begin{mathpar}
\inferrule*[right=TV-Set] {
  \inferrule*[right=(1)] { } {\Delta[\Pi'[p]][\ell] = \ptp}
  \and
  \inferrule* { } {\ell \in \Pi'[p].\mf{labs}}
} {
  \tval{\Gamma}{\Pi'}{\vset{\ell}{p}}{\tset{\ptp}}\ (4)
}
\end{mathpar}
\item We apply \thref{thm:wkpiv} (weakening) to (2) to get (5) that
$\tval{\Gamma}{\Pi'}{v}{\ptp}$. Then \thref{thm:subbod} (substitution),
applied twice to (3) using (4) and (5),
yields $\tbod{\Gamma}{\Pi'}{\hat{e}[v/x, \vset{\ell}{p}/x_\mf{set}]}{\C}{\Sigma'}$,
satisfying conclusion \emph{(i)}.
\item To show conclusion \emph{(ii)} given premise \emph{(ii)},
it suffices to show that\\
$\dom(m'[p].\mf{st}) = \Pi'[p].\mf{labs}$
(which is immediate from choice of $\Pi'$), and that
$\tval{\Gamma}{\Pi}{m[p].\mf{st}[\ell]}{\Delta[\C][\ell]}$.
\item By construction of $m'$, we need to show
$\Gamma \semi \Pi \vdval v : \Delta[\C][\ell]$.
This is given by (1) and (2).
\item Finally, we need to show conclusion \emph{(iii)}, i.e.
$\Pi'[p].\mf{labs} \cup \Sigma' = \Pi[p].\mf{labs} \cup \Sigma$.
Note that $\Sigma = \Sigma' \cup \{\ell\}$
and $\Pi'[p].\mf{labs} = \Pi[p].\mf{labs} \cup \{\ell\}$.
The conclusion follows from the equality:
$$(\Pi[p].\mf{labs} \cup \{\ell\}) \cup \Sigma' 
= \Pi[p].\mf{labs} \cup (\Sigma' \cup \{\ell\})$$
\end{itemize}

\tb{Case} \ts{EB-StS}: Impossible,
since the judgment requires that $\phi=\mf{Succ}$.
\end{Proof}
\end{lemma}

\begin{lemma}[Progress of bodies, successor phase.]
\label{thm:progbods}
If \begin{enumerate}[label=(\roman*)]
\item $\tbod{\cdot}{\Pi}{\hat{e}}{\Pi[p]}{\Sigma}$,
\item $\tmem{\cdot}{\Pi}{m}$, and
\item $\Sigma \subseteq \Pi[p].\mf{labs}$,
\end{enumerate}
then either:
\begin{enumerate}[label=(\roman*)]
\item $\hat{e} = \eret{v}$, or
\item there exists some $\hat{e}'$ such that
$\bstep{m}{\hat{e}}{\mf{Succ}}{p}{\hat{e}'}{m}$.
\end{enumerate}
\begin{Proof}
By induction over the derivation of the body typing judgment.

\tb{Case} \ts{TB-Lift}: \begin{mathpar}
\inferrule*[right=TB-Lift] {
  \inferrule* {(1)} {\texp{\cdot}{\Pi}{e}{\tview}{\epsilon}}
} {
  \tbod{\cdot}{\Pi}{e}{\C}{\emptyset}
}
\end{mathpar}
\begin{itemize}
\item By \thref{thm:progexp} (expression progress) on (1), we have two subcases:
either $e=\eret{v}$ (and we are done by conclusion (\emph{i})), 
or we have some $e'$ and $m'$ such that
(2) $\estep{m}{e}{e'}{m'}$.
\item By \thref{thm:pureexp} (purity) on (1) and (2), we get (3) that
$\estep{m}{e}{e'}{m}$.
\item We derive conclusion \emph{(ii)} as follows:
\begin{mathpar}
\inferrule*[right=EB-Lift] {
  \inferrule*[right=(3)] { } {\estep{m}{e}{e'}{m}}
} {
  \bstep{m}{e}{\mf{Succ}}{p}{e'}{m}
}
\end{mathpar}
\end{itemize}

\tb{Case} \ts{TB-Let}: \begin{mathpar}
\inferrule*[right=TB-Let] {
  \inferrule* {(1)} {\texp{\cdot}{\Pi}{e}{\ptp}{\epsilon}}
  \and
  \tbod{[x \mapsto \ptp]}{\Pi}{\hat{e}}{\C}{\Sigma}
} {
  \tbod{\cdot}{\Pi}{\elet{x}{e}{\hat{e}}}{\C}{\Sigma}
}
\end{mathpar}\begin{itemize}
\item By \thref{thm:progexp} (expression progress) on (1), we have two subcases:
either (a) $e = \eret{v}$,
or (b) we have $\estep{m}{e}{e'}{m'}$.
\item In subcase (a), \ts{EB-Ret} gives
$\bstep{m}{\elet{x}{\eret{v}}{\hat{e}}}{\mf{Succ}}{p}{\hat{e}[v/x]}{m}$.
\item In subcase (b), by \thref{thm:pureexp} (purity) on (1) and (b),
we get (3) that $\estep{m}{e}{e'}{m}$.
We derive a step as follows:
\begin{mathpar}
\inferrule*[right=EB-Let] {
  \inferrule*[right=(3)] { } {\estep{m}{e}{e'}{m}}
} {
  \bstep{m}{\elet{x}{e}{\hat{e}}}{\mf{Succ}}{p}{\elet{x}{e'}{\hat{e}}}{m}
}
\end{mathpar}
\item Thus, in either subcase, we take a step (with an unchanged memory)
and satisfy conclusion \emph{(ii)}.
\end{itemize}

\tb{Case} \ts{TB-State}: \begin{mathpar}
\inferrule*[right=TB-State] {
  \Delta[\C][\ell] = \ptp
  \and
  \tval{\cdot}{\Pi}{v}{\ptp}
  \\
  \tbod{\cdot[x \mapsto \ptp, x_\mf{set} \mapsto \tset{\ptp}]}
  {\Pi}{\hat{e}}{\C}{\Sigma}
} {
  \tbod{\cdot}{\Pi}{\estate{x}{\ell}{v}{\hat{e}}}{\C}{\Sigma \cup \{\ell\}}
}
\end{mathpar}
\begin{itemize}
\item We want to apply step rule \ts{EB-StS}, which says:
$$\bstep{m}{\estate{x}{\ell}{v}{\hat{e}}}{\mf{Succ}}{p}
{\hat{e}[v'/x, \vset{\ell}{p}/x_\mf{set}]}{m}$$
where $v' = m[p].\mf{st}[\ell]$.
It suffices to show that $v'$ is well-defined.
\item By premise \emph{(iii)} and the typing rule conclusion, 
we have that $\ell \in \Pi[p].\mf{labs}$.
Then by premise \emph{(ii)}, we get that $\ell \in \dom(m[p].\mf{st})$,
i.e. $v'$ is well-defined and we are done.
\end{itemize}
\end{Proof}
\end{lemma}

\tb{Case} \ts{EB-StI}: Impossible,

In subcase (b), by \thref{thm:pureexp} (purity) on (3), we get
(4) that $\estep{\cdot}{e}{e'}{\cdot}$.
We derive a step as follows, satisfying conclusion \emph{(ii)}:

\end{itemize}

\end{Proof}
\end{lemma}

\begin{lemma}[Preservation of programs.]
\label{thm:prezprog}
If
\begin{enumerate}[label=(\roman*)]
\item $\tprog{\Gamma}{P}{\Omega}$,
\item $\tdef{\Gamma}{\Omega_D}{\D}$, and
\item $\pstep{\D}{P}{P'}{\D'}$,
\end{enumerate}
then there exist some $\Omega' \subseteq \Omega$
and $\Omega_D' \supseteq \Omega_D$ such that:
\begin{enumerate}[label=(\roman*)]
\item $\tprog{\Gamma}{P'}{\Omega'}$,
\item $\tdef{\Gamma}{\Omega_D'}{\D'}$, and
\item $\Omega' \cup \Omega_D' = \Omega \cup \Omega_D$.
\end{enumerate}

\begin{Proof}
Unless stated otherwise, let $\Omega' = \Omega$
and $\Omega_D' = \Omega_D$
(such that both conclusions \emph{(ii)} and \emph{(iii)}
are trivial when $\D' = \D$.)

Again note that the empty memory is well-typed (only) by the empty path context,
i.e. we have (1) that $\tmem{\cdot}{\cdot}{\cdot}$.

We proceed by induction on the derivation of the step judgment \emph{(iii)}.

\tb{Case} \ts{EP-Lift}: We have \begin{mathpar}
\inferrule*[right=EP-Lift] {
  \inferrule* {(2)} {\estep{\cdot}{e}{e'}{\cdot}}
} {
  \pstep{\D}{e}{e'}{\D}
}
\and
\inferrule*[right=TP-Lift] {
  \inferrule* {(3)} {\texp{\Gamma}{\cdot}{e}{\tview}{\epsilon}}
} {
  \tprog{\Gamma}{e}{\emptyset}
}
\end{mathpar}
\begin{itemize}
\item \thref{thm:prezexp} (expression preservation)
on (2), (3), and the empty memory (1) gives us (4) that 
$\texp{\Gamma}{\cdot}{e'}{\tview}{\epsilon}$.
\item We derive conclusion \emph{(i)} as follows:
\begin{mathpar}
\inferrule*[right=TP-Lift] {
  \inferrule*[right=(3)] { } {\texp{\Gamma}{\cdot}{e'}{\tview}{\epsilon}}
} {
  \tprog{\Gamma}{e'}{\emptyset}
}
\end{mathpar}
\end{itemize}

\tb{Case} \ts{EP-Let}: We have \begin{mathpar}
\inferrule*[right=EP-Let] {
  \inferrule* {(2)} {\estep{\cdot}{e}{e'}{\cdot}}
} {
  \pstep{\D}{\elet{x}{e}{P}}{\elet{x}{e'}{P}}{\D}
}
\and
\inferrule*[right=TP-Let] {
  \inferrule* {(3)} {\texp{\Gamma}{\cdot}{e}{\ptp}{\epsilon}}
  \and 
  \inferrule* {(4)} {\tprog{\Gamma[x \mapsto \ptp]}{P}{\Omega}}
} {
  \tprog{\Gamma}{\elet{x}{e}{P}}{\Omega}
}
\end{mathpar}
\begin{itemize}
\item \thref{thm:prezexp} (expression preservation)
on (2), (3), and the empty memory (1) gives us (5) that 
$\texp{\Gamma}{\cdot}{e'}{\ptp}{\epsilon}$.
\item We derive conclusion \emph{(i)} as follows:
\begin{mathpar}
\inferrule*[right=TP-Let] {
  \inferrule*[right=(5)] { } {\texp{\Gamma}{\cdot}{e'}{\ptp}{\epsilon}}
  \and 
  \inferrule*[right=(4)] { } {\tprog{\Gamma[x \mapsto \ptp]}{P}{\Omega}}
} {
  \tprog{\Gamma}{\elet{x}{e'}{P}}{\Omega}
}
\end{mathpar}
\end{itemize}

\tb{Case} \ts{EP-Ret}: We have
$\D \semi \elet{x}{\eret{v}}{P} \mapsto P[v/x] \semi \D$ and 
\begin{mathpar}
\inferrule*[right=TP-Let] {
  \inferrule*[right=TE-Ret] {
    \inferrule* {(2)} {\tval{\Gamma}{\cdot}{v}{\ptp}}
  } {
    \texp{\Gamma}{\cdot}{\eret{v}}{\ptp}{\epsilon}
  }
  \and 
  \inferrule* {(3)} {\tprog{\Gamma[x \mapsto \ptp]}{P}{\Omega}}
} {
  \tprog{\Gamma}{\elet{x}{\eret{v}}{P}}{\Omega}
}
\end{mathpar}
\begin{itemize}
\item Conclusion \emph{(i)}, that $\tprog{\Gamma}{P[v/x]}{\Omega}$,
follows directly from \thref{thm:subprog} (substitution) on (2) and (3).
\end{itemize}

\tb{Case} \ts{EP-Def}: We have
$\pstep{\D}{\elet{\C(x)}{\hat{e}}{P}}{P}{\D[\C \mapsto x.\hat{e}]}$
and
\begin{mathpar}
\inferrule*[right=TP-Def] {
  \inferrule* {(2)} {
    \tbod{\Gamma[x \mapsto \Delta[\C]]}{\cdot}{\hat{e}}{\C}{\dom(\Delta[\C])}
  }
  \and
  \inferrule* {(3)} {\tprog{\Gamma}{P}{\hat{\Omega}}}
} {
  \tprog{\Gamma}{\elet{\C(x)}{\hat{e}}{P}}{\hat{\Omega} \cup \{\C\}}
}
\end{mathpar}
\begin{itemize}
\item Choose $\Omega' = \hat{\Omega}$ , 
such that $\Omega = \Omega' \cup \{\C\}$.
Clearly $\Omega' \subseteq \Omega$, and furthermore,
conclusion \emph{(i)} is immediate from (3).
\item Also choose $\Omega_D' = \Omega_D \cup \{C\}$.
Now in order to show conclusion \emph{(ii)} given premise \emph{(ii)},
it suffices to show that 
$\tbod{\Gamma[x \mapsto \Delta[\C]]}{\cdot}{\hat{e}}{\C}{\dom(\Delta[\C])}$.
This is immediate from (2).
\item Finally, we need to show conclusion \emph{(iii)}, i.e.
$\Omega' \cup \Omega_D' = \Omega \cup \Omega_D$.
The conclusion follows from definitions of $\Omega$ and $\Omega_D'$:
$$\Omega' \cup (\Omega_D \cup \{\C\}) = (\Omega' \cup \{\C\}) \cup \Omega_D$$
\item All conclusions hold, and we are done.
\end{itemize}
\end{Proof}
\end{lemma}

\section{Small-Step Runtime}

\subsection{Runtime Syntax}

\begin{align*}
\text{Runtime values } u
&::= ()_\rho \mid []_\rho \mid u ::_\rho u
\mid \elit{v} \mid \enode{id}{v}{u}
\mid p\\
\text{Runtime expressions } \rho
&::= ()_\rho \mid []_\rho \mid \rho ::_\rho \rho
\mid \elit{v} \mid \enode{id}{v}{\rho}
\mid p\\
&\quad\mid \rho \semi \rho \mid
\einit{v} \mid \einits{v} \mid \escan{v} \mid \escans{v}\\
&\quad\mid \echk{\rho} \mid \echks{\rho}
\mid \esearch{id}{\rho} \mid \esearches{id}{\rho}\\
&\quad\mid \eeval{\phi}{p}{\hat{e}} \mid \enew{\phi}{p}{\rho}
\mid \ehdl{e} \mid \eprog{P}\\
\text{Evaluation contexts } \E
&::= [\cdot] \mid \enode{id}{v}{\E} \mid \E ::_\rho \rho
\mid u ::_\rho \E \mid \E \semi \rho\\
&\quad\mid \enew{\phi}{p}{\E}
\mid \echk{\E} \mid \echks{\E}\\
&\quad\mid \esearch{id}{\E} \mid \esearches{id}{\E}\\
\text{Runtime type } \rtp &::= \tone_\rho \mid \mf{Tree}
\mid [\rtp]_\rho
\end{align*}

\tb{\emph{Configurations.}} Our configurations are triples of the form
$\trip{\D}{m}{\rho}$.

\tb{\emph{Subscripts.}} 
Some forms are subscripted with $\rho$ to indicate a
\emph{syntactic distinction} between them and their program-level
counterparts. (This $\rho$ is just a marker, not a runtime expression.)

In general, however, we omit these markers when doing so causes no ambiguity.

\tb{\emph{Phase Types.}}
We associate each phase $\phi$ with a runtime type $\rtp$ as follows:
\[
\rtp(\phi) = \begin{cases}
    \mf{Tree} & \phi=\mf{Init}\\
    \tone & \phi=\mf{Succ}
\end{cases}
\]

\tb{\emph{Constants.}} We distinguish three constants:
\begin{itemize}
\item $p_0$, the ``root path'' in memory,
\item $\C_0$, the (dummy) component name associated with that path, and
\item $x_0$, the variable name bound in that component's body.
\end{itemize}
We assume these constants are \emph{globally fresh},
i.e. they do not conflict with any other names in the program
or that are ``generated'' at runtime, e.g. by the premise ``$p$ fresh''.

\subsection{Runtime-Level Proofs}

\subsubsection{Lemmas}

\begin{proposition}[Canonical forms, runtime.]
\label{thm:canonrt}
If $\trho{\cdot}{\Pi}{\Omega}{u}{\rtp}$, then:
\begin{itemize}
\item if $\rtp=\tone$, then $u=()$.
\item if $\rtp=\ttree$, then either $u = \elit{v}$,
$u = \enode{id}{v}{u}$, or $u = p$.
\item if $\rtp=[\rtp']$, then either $u=[]$ or $u = u :: us$.
\end{itemize}
\end{proposition}

\subsubsection{Soundness}

\begin{lemma}[Progress of configurations.]
\label{thm:progcfg}
If $\tcfg{\cdot}{\Pi}{\Omega}{\D}{m}{\rho}{\rtp}$, we then
have either
\begin{enumerate}[label=(\roman*)]
\item $\rho = u$, or
\item there exists some $\trip{\D'}{m'}{\rho'}$ such that
$\cstep{\D}{m}{\rho}{\D'}{m'}{\rho'}$.
\end{enumerate}

\begin{proof}
The premise gives us the following three sub-premises:
\begin{enumerate}[label=(\roman*)]
\item $\tdef{\cdot}{\Omega}{\D}$,
\item $\tmem{\cdot}{\Pi}{m}$, and
\item $\trho{\cdot}{\Pi}{\Omega}{\rho}{\rtp}$.
\end{enumerate}

Thus, to apply the inductive hypothesis to
a configuration $\trip{\D}{m}{\hat{\rho}}$, by premises (i) and (ii)
it suffices to show that $\trho{\cdot}{\Pi}{\Omega}{\hat{\rho}}{\rtp}$.

We abbreviate $\trho{\cdot}{\Pi}{\Omega}{\rho}{\rtp}$
to $\trhos{\cdot}{\Pi}{\rho}{\rtp}$ when $\Omega$ is irrelevant.

We proceed by induction over the derivation of premise (iii).

\tb{Cases} \ts{TR-Unit}, \ts{TR-Nil}, \ts{TR-Lit},
\ts{TR-Path}: All these rules only type values,
and so conclusion \emph{(i)} holds easily.

\tb{Case} \ts{TR-Node}: \begin{mathpar}
\inferrule*[right=TR-Node] {
  \tval{\cdot}{\Pi}{v}{[\tattr]}
  \and
  \inferrule* {(1)} {\trhos{\cdot}{\Pi}{\rho}{[\ttree]}}
} {
  \trhos{\cdot}{\Pi}{\enode{id}{v}{\rho}}{\ttree}
}
\end{mathpar}
\begin{itemize}
\item By inductive hypothesis on (1), we either have $\rho=u$
or (2) $\cstep{\D}{m}{\rho}{\D'}{m'}{\rho'}$.
In the former case, $\enode{id}{v}{u}$ is also a value
and we are done by conclusion \emph{(i)}.
\item In the latter case, we apply rule \ts{ER-Ctx}
with $\E = \enode{id}{v}{[\cdot]}$ as follows to get a step,
satisfying conclusion \emph{(ii)}:
\begin{mathpar}
\inferrule*[right=ER-Ctx] {
  \inferrule*[right=(2)] { } {\cstep{\D}{m}{\rho}{\D'}{m'}{\rho'}}
} {
  \cstep{\D}{m}{\enode{id}{v}{\rho}}{\D'}{m'}{\enode{id}{v}{\rho'}}
}
\end{mathpar}
\end{itemize}

\tb{Case} \ts{TR-Cons}: \begin{mathpar}
\inferrule*[right=TR-Cons] {
  \inferrule* {(1)} {\trhos{\cdot}{\Pi}{\rho}{\rtp}}
  \and
  \inferrule* {(2)} {\trhos{\cdot}{\Pi}{\rho s}{[\rtp]}}
} {
  \trhos{\cdot}{\Pi}{\rho :: \rho s}{[\rtp]}
}
\end{mathpar}
\begin{itemize}
\item By inductive hypothesis on (1), we either have 
(a) that $\cstep{\D}{m}{\rho}{\D'}{m'}{\rho'}$
or (b) that $\rho=u$.
\item In subcase (a), we apply rule \ts{ER-Ctx}
with $\E = [\cdot] :: \rho s$ as follows to get a step,
satisfying conclusion \emph{(ii)}:
\begin{mathpar}
\inferrule*[right=ER-Ctx] {
  \inferrule*[right=\tn{(a)}] { } {\cstep{\D}{m}{\rho}{\D'}{m'}{\rho'}}
} {
  \cstep{\D}{m}{\rho :: \rho s}{\D'}{m'}{\rho' :: \rho s}
}
\end{mathpar}
\item In subcase (b), we apply the inductive hypothesis to (2)
to get two more subcases,
that either (b-i) $\cstep{\D}{m}{\rho s}{\D'}{m'}{\rho s'}$,
or (b-ii) that $\rho s = us$.
\item In subcase (b-i), we derive a step via \ts{ER-Ctx}
with $\E = u :: [\cdot]$ as follows:
\begin{mathpar}
\inferrule*[right=ER-Ctx] {
  \inferrule*[right=\tn{(b-i)}] { } {\cstep{\D}{m}{\rho s}{\D'}{m'}{\rho s'}}
} {
  \cstep{\D}{m}{u :: \rho s}{\D'}{m'}{u :: \rho s'}
}
\end{mathpar}
\item In subcase (b-ii), we have that $\rho :: \rho s = u :: us$,
i.e. we have a value and we are done by conclusion \emph{(i)}.
Thus, all subcases satisfy some conclusion, and we are done.
\end{itemize}

\tb{Case} \ts{TR-Seq}: \begin{mathpar}

\begin{lemma}[Preservation of configurations.]
\label{thm:prezcfg}
If \begin{enumerate}[label=(\roman*)]
\item $\tcfg{\Gamma}{\Pi}{\Omega}{\D}{m}{\rho}{\rtp}$ and
\item $\cstep{\D}{m}{\rho}{\D'}{m'}{\rho'}$,
\end{enumerate}
then, for some $\Pi' \rpo \Pi$ and $\Omega' \supseteq \Omega$,
we have $\tcfg{\Gamma}{\Pi'}{\Omega'}{\D'}{m'}{\rho'}{\rtp}$.

\begin{Proof}
Premise \emph{(i)} gives us the following three sub-premises:
\begin{enumerate}[label=(i-\alph*)]
\item $\tdef{\Gamma}{\Omega}{\D}$,
\item $\tmem{\Gamma}{\Pi}{m}$, and
\item $\trho{\Gamma}{\Pi}{\Omega}{\rho}{\rtp}$.
\end{enumerate}

Likewise, we want to show the following three sub-conclusions:
\begin{enumerate}[label=(\roman*)]
\item $\tdef{\Gamma}{\Omega'}{\D'}$,
\item $\tmem{\Gamma}{\Pi'}{m'}$, and
\item $\trho{\Gamma}{\Pi'}{\Omega'}{\rho'}{\rtp}$.
\end{enumerate}

Unless otherwise specified, we choose $\Omega'=\Omega$
and $\Pi'=\Pi$, such that conclusions (i)
and (ii) are immediate from premises (i-a) and (i-b)
when $\D'=\D$ and $m'=m$, respectively.

As with progress, we work with sub-judgment
$\trhos{\Gamma}{\Pi}{\rho}{\rtp}$ unless $\Omega$ is relevant.

We proceed by induction over the step judgment in premise \emph{(ii)}.

\tb{Case} \ts{ER-Ctx}: We have \begin{mathpar}
\inferrule*[right=ER-Ctx] {
  \inferrule* {(1)} {\cstep{\D}{m}{\rho}{\D'}{m'}{\rho'}}
} {
  \cstep{\D}{m}{\E[\rho]}{\D'}{m'}{\E'[\rho]}
}
\end{mathpar}

\item \hl{TO-DO}... needs lemmas (e.g. replacement?)
\end{itemize}

\section{Linear-Typed Runtime}

We define $\Theta=\{\num{p}\}$ to be a \emph{linear} context
of paths. As a well-formedness condition,
$\Theta$ should not contain duplicates, and 
context composition is defined in the usual linear way.

\subsection{Memory}

We define $\tpi{\Gamma}{\Pi}{\Theta}{\pi}{p}$ to hold if,
for $\pi.\mf{spec}=\pair{\C}{v}$:
\begin{itemize}
\item $\Pi[p].\mf{spec}=\C$,
\item $\tval{\Gamma}{\Pi}{v}{\Delta[\C]}$,
\item $\dom(\pi.\mf{st}) = \Pi[p].\mf{labs}$,
\item for all $\ell \in \dom(\pi.\mf{st})$,
we have $\tval{\Gamma}{\Pi}{\pi[p].\mf{st}[\ell]}{\Delta[\C][\ell]}$,
\item if $\Pi[p].\mf{tree} = \tt$, then
$\trhol{\Gamma}{\Pi}{\Theta}{\pi.\mf{tree}}{\ttree}$,
\item if instead $\Pi[p].\mf{tree} = \ff$, then $\Theta=\cdot$, and
\item $p \notin \Theta$.
\end{itemize}
Above, $\Theta$ is the set of paths ``owned'' by $p$.

We define $\tmeml{\Gamma}{\Pi}{\Theta}{m}$ to hold if
there is some series of $\Theta_i$ with $\num{\Theta_i}=\Theta$,
and, for all $p_i \in \dom(m)$,
$\tpi{\Gamma}{\Pi}{\Theta_i}{m[p_i]}{p_i}$.
We also require that $p_0 \notin \Theta$.

Above, $\Theta$ is the set of all ``owned'' paths in the memory.

to the expression in question),

\subsection{Type Soundness, Revisited}

Note that
$\tcfgl{\Gamma}{\Pi}{\Theta}{\Omega}{\D}{m}{\rho}{\rtp}$
implies our old judgment
$\tcfg{\Gamma}{\Pi}{\Omega}{\D}{m}{\rho}{\rtp}$,
so progress follows from our old \thref{thm:progcfg}.

\begin{lemma}[Preservation of configurations, linear extension.]
\label{thm:prezcfgl}
If\\ $\tcfgl{\Gamma}{\Pi}{\Theta}{\Omega}{\D}{m}{\rho}{\rtp}$
and $\cstep{\D}{m}{\rho}{\D'}{m'}{\rho'}$,
then there exist some $\Pi' \rpo \Pi$, $\Theta' \supseteq \Theta$,
and $\Omega' \supseteq \Omega$ such that
$\tcfgl{\Gamma}{\Pi'}{\Theta'}{\Omega'}{\D'}{m'}{\rho'}{\rtp}$.

\begin{Proof}
We detail only those cases which significantly differ from
\thref{thm:prezcfg} via the introduction of $\Theta$,
or are otherwise noteworthy.
The other cases are straightforward by choosing $\Theta' = \Theta$.

\tb{Case} \ts{ER-Ctx}: Nontrivial, and \hl{depends on our formulation}
of this case in \thref{thm:prezcfg}.

In particular, for an expression like $\einit{\pair{\C}{v}} \semi \rho$,
we'll type each subexpression via $\Theta_1$ and $\Theta_2$ respectively,
where $\Theta=\Theta_1,\Theta_2$.
When we step $\einit{\pair{\C}{v}}$, we extend $\Theta_1$ by some 
``fresh'' $p$, but this $p$ must be fresh \emph{with respect to} $\Theta$,
not just $\Theta_1$, in order for the result $\Theta,p$ to be well-formed.

This property should follow when we say that ``$p$ fresh'' means $p \notin \dom(m)$,
which means $p \notin \dom(\Pi)$, and since $\Theta \subset \dom(\Pi)$,
we get $p \notin \Theta$. However, I don't know \hl{how to formalize this idea
across all evaluation contexts}.

\tb{Case} \ts{ER-ISpec}: We have
\begin{mathpar}
\inferrule*[right=ER-ISpec] {
  p\ \mf{fresh}
  \and 
  \D[\C]=x.\hat{e}
  \\
  m'=m[p \mapsto \{
    \mf{spec}=\pair{\C}{v}, 
    \mf{st}: \cdot, 
    \mf{tree}: \cdot, 
    \mf{rerender}: \mf{ff}
  \}]
} {
  \cstep{\D}{m}{\einit{\pair{\C}{v}}}{\D}{m'}{\eeval{\mf{Init}}{p}{\hat{e}[v/x]}}
}
\and
\inferrule*[right=TL-Init] {
  \inferrule*[right=TV-Spec] {
    \inferrule* {(1)} {\tval{\Gamma}{\Pi}{v}{\Delta[\C]}}
  } {
    \tval{\Gamma}{\Pi}{\pair{\C}{v}}{\tview}
  }
} {
  \trhol{\Gamma}{\Pi}{\cdot}{\einit{\pair{\C}{v}}}{\ttree}
}
\end{mathpar}
\begin{itemize}
\item In (only) this case, we extend $\Theta$ with our fresh $p$,
i.e. $\Theta'_\rho = \{p\}$. Via the same choice of $\Pi'$, 
we derive (2) and (3) in the same way as \thref{thm:prezcfg},
and the conclusion follows just as before:
\end{itemize}
\begin{mathpar}
\inferrule*[right=TL-EvalI] {
  \inferrule*[right=(2)] { } {
    \tbod{\Gamma}{\Pi'}{\hat{e}[v/x]}{\C}{\Sigma}
  }
  \
  \inferrule*[right=(3)] { } {
    \dom(\Delta[\C]) = \Sigma \cup \Pi'[p].\mf{labs}
  }
} {
  \trhol{\Gamma}{\Pi'}{p}{\eeval{\mf{Init}}{p}{\hat{e}[v/x]}}{\ttree}
}
\end{mathpar}

\tb{Case} \ts{ER-NewI}: We have
$\cstep{\D}{m}{\enew{\mf{Init}}{p}{u}}
{\D}{m'}{p}$
with $m'$ defined to be $m \mid m[p].\mf{tree} = u$,
and: \begin{mathpar}
\inferrule*[right=TL-NewI] {
  \inferrule* {(1)} {
    \trhol{\Gamma}{\Pi}{\Theta_\rho}{u}{\ttree}
  }
  \
  \inferrule* {(2)} {
    \Pi[p].\mf{labs} = \dom(\Delta[\Pi[p]])
  }
} {
  \trhol{\Gamma}{\Pi}{\Theta_\rho,p}{\enew{\mf{Init}}{p}{u}}{\ttree}
}
\end{mathpar}
\begin{itemize}
\item We choose $\Theta'=\Theta$, but we \emph{redistribute} $\Theta'$
between the memory and the expression,
such that $\Theta_m' = \Theta_m,\Theta_\rho$ and $\Theta_\rho'=\{p\}$.
\item Within the memory, we distribute $\Theta_m$ across paths
exactly as before, and we let $\Theta_\rho$ correspond to $p$.
\item Finally, choose $\Pi' = \Pi \mid \Pi[p].\mf{tree} = \tt$.
By (2), we already have $p \in \dom(\Pi)$,
so to show conclusion (ii), it suffices to show
that $\trhol{\Gamma}{\Pi}{\Theta_\rho}{u}{\ttree}$, which is given by (1).
\item Conclusion (iii) follows from the derivation:
\begin{mathpar}
\inferrule*[right=TL-Path] {
  \inferrule*[right=TI-Path] {
    \inferrule* { } {\Pi'[p].\mf{tree} = \tt}
    \and
    \inferrule*[right=(2)] { } {
      \Pi'[p].\mf{labs} = \dom(\Delta[\Pi'[p]])

    \trhoi{\Gamma}{\Pi}{p}{\ttree}

  \trhol{\Gamma}{\Pi}{p}{p}{\ttree}

\end{mathpar}
\end{itemize}

\tb{Case} \ts{ER-NewS}: We have
$\cstep{\D}{m}{\enew{\mf{Succ}}{p}{u}}{\D}{m'}{()}$,
where $m'$ is defined as $m \mid m[p].\mf{tree} = u$,
and: \begin{mathpar}
\inferrule*[right=TL-New] {
  \inferrule* {(1)} {\trhol{\Gamma}{\Pi}{\Theta_\rho}{u}{\ttree}}
  \and
  \Pi[p].\mf{labs} = \dom(\Delta[\Pi[p]])

  \trhol{\Gamma}{\Pi}{\Theta_\rho}{\enew{\mf{Succ}}{p}{u}}{\tone}

\end{mathpar}
\begin{itemize}
\item We redistribute $\Theta$ in an analogous way to the prior case,
except that $\Theta_\rho'=\emptyset$. The memory is thus well-typed
via an analogous argument,
and conclusion (iii) follows from \ts{TR-Unit}
(under an empty $\Theta_\rho'$).
\end{itemize}

\end{Proof}
\end{lemma}

\subsection{Realization}

We define the $\mf{unfold}$ meta-function to ``realize'' a memory into
a concrete tree structure as follows:
\begin{align*}
\mf{unfold}(m, \elit{v}) &= \elit{v}\\
\mf{unfold}(m, \enode{id}{v}{u}) &= \enode{id}{v}{\mf{unfold}\ u}\\
\mf{unfold}(m, p) &= \mf{unfold}(m[p].\mf{tree})\\
\mf{unfold}(m, []) &= []\\
\mf{unfold}(m, u :: us) &= \mf{unfold}\ u :: \mf{unfold}\ us\\
\end{align*}

\begin{theorem}[Termination of realization.]
\label{thm:realterm}
$\tcfgl{\cdot}{\Pi}{\Theta}{\Omega}{\D}{m}{u}{\ttree}$
implies that $\mf{unfold}(m, u)$ terminates.

\begin{Proof}
\hl{Incomplete}: not sure how to phrase this idea formally.

We treat the equational definition of $\mf{unfold}$
as a series of (left-to-right) \emph{rewriting rules}
in order to reason about individual steps.

We define a measure $\mc{M}$ \emph{on the typing judgment}
in $\mathbb{P}(\mf{Path}) \times \mathbb{N}$,
such that the first component is $\Theta$ and
the second is the (syntactic) size of $u$.
Since $\Theta$ must be finite, we have
a well-founded lexicographic order given by
$\subset$ and $<$ on respective components.
To prove termination, it suffices to show that
\begin{enumerate}[label=(\roman*)]
\item $\mf{unfold}(m, u)$ is well-defined 
for any $m$ and $u$ (which are well-typed in the configuration),
\item any subcalls to $\mf{unfold}(m, u')$ are well-typed
by some $\Theta' \subseteq \Theta$,
i.e.\\$\tcfgl{\cdot}{\Pi}{\Theta'}{\Omega}{\D}{m}{u'}{\ttree}$, and
\item the measure $\mc{M}$ on the typing derivations of those
subcalls is strictly decreasing.
\end{enumerate}

We case-analyze on the canonical forms of $u$.
By definition of $\mf{unfold}$,
clearly $u$'s size decreases in every case
\emph{except} for when $u=p$.
We choose $\Theta'=\Theta$, and clearly
$\mc{M}$ decreases. The typing conclusion
is obvious from the premise.

When $u=p$, we know by well-typedness that 
$m[p].\mf{tree}$ is well-defined, and that $\Theta_\rho=p$.
By construction, $p \notin \Theta_m$.
Choose $\Theta' = \Theta_m$;
clearly $\Theta' \subset \Theta$...
\end{Proof}
\end{theorem}

\subsection{Abbreviations}

We use meta-notation to define type operators, and we
define data types and their constructors via Haskell-like syntax
for GADTs when convenient. 
For instance, we might declare the $\mf{List}$ type as follows:

\begin{lstlisting}[language=Haskell,mathescape]
data List a where
  Nil :: List a
  Cons :: a -> List a -> List a
\end{lstlisting}
which desugars to the following type:
$$\mf{List}\ A = \mu \alpha. 1 + (A \times \alpha)$$
and the following constructors:
\begin{align*}
\mf{Nil} &= \einl{()}\\
\mf{Cons} &= \lambda hd.\; \lambda tl.\; \einr{\pair{hd}{tl}}
\end{align*}

(We also say $[A]$ to mean $\mf{List}\ A$.)

Other abbreviations and definition relevant to our translation are as follows:

\begin{lstlisting}[language=Haskell,mathescape]
data Attr where
  Attr :: Str -> Str -> Attr
  OnClick :: (1 -> 1) -> Attr
\end{lstlisting}

\begin{lstlisting}[language=Haskell,mathescape]
data View where
  Text :: Str -> View
  Tag :: [Attr] -> [View] -> View
  Spec :: $\forall$ a. (a -> (Ref Bool * 1 -> View) * a) -> View
\end{lstlisting}
(Note that the LHS of the $\mf{Spec}$ constructor is \emph{existential}.)

\begin{lstlisting}[language=Haskell,mathescape]
data Tree where
  Lit :: Str -> View
  Node :: ID -> [Attr] -> [View] -> View
  Path :: $\forall$ a. (a -> (Ref Bool * 1 -> View) * a)
    -> (1 -> View) -> (Ref Tree) -> (Ref Bool) -> Tree
\end{lstlisting}
